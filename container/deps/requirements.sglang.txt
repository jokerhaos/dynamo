# Third-party Python dependencies for the SGLang runtime image. Installed with
# --force-reinstall --no-deps so the upstream SGLang dependency stack remains
# otherwise unchanged. codec-bearing video-DECODE wheels (av, decord, opencv)
# are removed separately in sglang_runtime.Dockerfile; the in-tree VP9 ffmpeg +
# imageio-ffmpeg below are kept for the video-generation encode path.
# torchcodec is the ONE decode carrier we keep (restored below): SGLang's
# sglang.srt.utils.video_decoder picks it as its primary backend and has no
# working video-input path without it. Keep this pin in sync with the SGLang
# pyproject the base image builds from (v0.5.20 pins torchcodec==0.15.0).

# Install imageio-ffmpeg from source so no prebuilt ffmpeg binary lands in
# <site-packages>/imageio_ffmpeg/binaries/. IMAGEIO_FFMPEG_EXE (set in
# sglang_runtime.Dockerfile) points imageio at the in-tree VP9 CLI copied from
# wheel_builder. The video-generation handler encodes with libvpx-vp9.
--no-binary imageio-ffmpeg

blake3>=1.0.0,<2.0.0  # Dynamo SGLang multimodal request handlers import blake3 at startup
imageio-ffmpeg>=0.6.0  # binary skipped per --no-binary directive above
# The upstream SGLang image bundles pillow at an older patch release; floor it
# so the install refreshes it to the current patch line.
pillow>=12.3.0,<13  # bounded: this file installs with --force-reinstall
# NVDEC hardware decode for H.264/H.265 video input (dynamo common.multimodal.
# nvdec_decoder), restoring those codecs after the video-DECODE purge in
# sglang_runtime.Dockerfile. Pin >=2.2.0: it bundles only libavutil + libavformat
# (container demux, no software codec); older 2.0.x bundles a full FFmpeg incl.
# libavcodec that the codec gate rejects. Links libnvcuvid at runtime; requires
# NVIDIA_DRIVER_CAPABILITIES to include "video" (set in sglang_runtime.Dockerfile).
PyNvVideoCodec>=2.2.0
# Primary software video-decode backend for SGLang's video input path
# (video_decoder.py: torchcodec preferred, decord as fallback). The wheel
# vendors its own (LGPL-2.1-or-later) FFmpeg carrying software H.264/H.265
# decoders -- accepted for this internal image; the runtime keep is excluded
# from the codec purge in sglang_runtime.Dockerfile and from the bundled-
# libavcodec scan in tests/dependencies/test_no_software_video_codecs.py.
torchcodec==0.15.0
zstandard==0.23.0
